import matplotlib.pyplot as plt
from sklearn.datasets import load_breast_cancer
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

X = StandardScaler().fit_transform(load_breast_cancer().data)
km = KMeans(n_clusters=2, random_state=42, n_init=10).fit(X)

plt.scatter(X[:,0], X[:,1], c=km.labels_,cmap='viridis', s=50)
plt.scatter(*km.cluster_centers_[:,:2].T, c='red', s=200,  marker='x', label='Centroids')
plt.title('K-Means - Breast Cancer'); plt.legend();
plt.show()
