import matplotlib.pyplot as plt
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier, plot_tree
from sklearn.metrics import accuracy_score, classification_report

d = load_breast_cancer()

Xtr, Xte, Ytr, Yte = train_test_split(d.data, d.target, test_size=0.2, random_state=42)
clf = DecisionTreeClassifier(random_state=42).fit(Xtr, Ytr)
predicted = clf.predict(Xte)

print("Accuracy Score:", accuracy_score(Yte, predicted))
print(classification_report(Yte, predicted, target_names=d.target_names))

plt.figure(figsize=(12,8))
plot_tree(clf, filled=True, feature_names=d.feature_names, class_names=["Malignant", "Benign"], rounded=True, proportion=True)
print("Sample:", "Benign" if clf.predict(Xte[:1])[0] else "Malignant")
